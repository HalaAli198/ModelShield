    chmod +x run.sh

    cat ~/gadget_substitution_results/summary.json

    grep "Verdict:" ~/gadget_substitution_results/gadget_substitution_log.txt

    echo "TP: $(ls ~/gadget_substitution_results/TP/ | wc -l)"
